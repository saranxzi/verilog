module decoder2x4 (
    input [1:0] A,
    output reg [3:0] Y
);

always @ (A) begin
    case (A)
        2'b00: Y = 4'b0001;
        2'b01: Y = 4'b0010;
        2'b10: Y = 4'b0100;
        2'b11: Y = 4'b1000;
        default: Y = 4'b0000;
    endcase
end

endmodule

module decoder2x4_tb;

reg [1:0] A;      
wire [3:0] Y;     

decoder2x4 uut (
    .A(A),
    .Y(Y)
);

initial begin
    $display("A  | Y");
    $monitor("%b | %b", A, Y);

    A = 2'b00; #10;
    A = 2'b01; #10;
    A = 2'b10; #10;
    A = 2'b11; #10;

    $finish;
end

endmodule
