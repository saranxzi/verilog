module mux_4x1_struct(input [3:0] in, input [1:0] sel, output out);
  wire n0, n1;
  wire y0, y1, y2, y3;

  not(n0, sel[0]);
  not(n1, sel[1]);

  and(y0, in[0], n1, n0);
  and(y1, in[1], n1, sel[0]);
  and(y2, in[2], sel[1], n0);
  and(y3, in[3], sel[1], sel[0]);

  or(out, y0, y1, y2, y3);
endmodule

module tb_mux_4x1_struct;
  reg [3:0] in;
  reg [1:0] sel;
  wire out;

  mux_4x1_struct uut (.in(in), .sel(sel), .out(out));

  initial begin
    in = 4'b1000; sel = 2'b00; #10;
    sel = 2'b01; #10;
    sel = 2'b10; #10;
    sel = 2'b11; #10;
    in = 4'b0110; sel = 2'b00; #10;
    sel = 2'b01; #10;
    $stop;
  end
endmodule
