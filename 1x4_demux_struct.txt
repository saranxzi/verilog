module demux_1x4_struct(input in, input [1:0] sel, output [3:0] out);
  wire n0, n1;

  not(n0, sel[0]);
  not(n1, sel[1]);

  and(out[0], in, n1, n0);
  and(out[1], in, n1, sel[0]);
  and(out[2], in, sel[1], n0);
  and(out[3], in, sel[1], sel[0]);
endmodule

module tb_demux_1x4_struct;
  reg in;
  reg [1:0] sel;
  wire [3:0] out;

  demux_1x4_struct uut (.in(in), .sel(sel), .out(out));

  initial begin
    in = 1'b1; sel = 2'b00; #10;
    sel = 2'b01; #10;
    sel = 2'b10; #10;
    sel = 2'b11; #10;
    in = 1'b0; sel = 2'b10; #10;
    $stop;
  end
endmodule
