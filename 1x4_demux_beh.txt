module demux_1x4_beh(input in, input [1:0] sel, output reg [3:0] out);
  always @ (in or sel) begin
    out = 4'b0000;
    case (sel)
      2'b00: out[0] = in;
      2'b01: out[1] = in;
      2'b10: out[2] = in;
      2'b11: out[3] = in;
    endcase
  end
endmodule

module tb_demux_1x4_beh;
  reg in;
  reg [1:0] sel;
  wire [3:0] out;

  demux_1x4_beh uut (.in(in), .sel(sel), .out(out));

  initial begin
    in = 1'b1; sel = 2'b00; #10;
    sel = 2'b01; #10;
    sel = 2'b10; #10;
    sel = 2'b11; #10;
    in = 1'b0; sel = 2'b01; #10;
    $stop;
  end
endmodule
