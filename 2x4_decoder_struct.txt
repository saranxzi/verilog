module decoder_2x4_structural(input [1:0] in, input en, output [3:0] out);
  wire n0, n1;
  not n_gate0(n0, in[0]);
  not n_gate1(n1, in[1]);
  and and_gate0(out[0], n1, n0, en);
  and and_gate1(out[1], n1, in[0], en);
  and and_gate2(out[2], in[1], n0, en);
  and and_gate3(out[3], in[1], in[0], en);
endmodule

module tb_decoder_2x4_structural;
  reg [1:0] in;
  reg en;
  wire [3:0] out;

  decoder_2x4_structural uut (.in(in), .en(en), .out(out));

  initial begin
    en = 0; in = 2'b00; #10;
    en = 1; in = 2'b00; #10;
    in = 2'b01; #10;
    in = 2'b10; #10;
    in = 2'b11; #10;
    en = 0; in = 2'b10; #10;
    $stop;
  end
endmodule
