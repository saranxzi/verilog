module pipo_beh(input clk, input reset, input [3:0] pin, output reg [3:0] pout);
  always @(posedge clk or posedge reset) begin
    if (reset)
      pout <= 4'b0000;
    else
      pout <= pin;
  end
endmodule

module tb_pipo_beh;
  reg clk, reset;
  reg [3:0] pin;
  wire [3:0] pout;

  pipo_beh uut (.clk(clk), .reset(reset), .pin(pin), .pout(pout));

  initial begin
    clk = 0;
    forever #5 clk = ~clk;
  end

  initial begin
    reset = 1; pin = 4'b0000; #10;
    reset = 0; pin = 4'b1010; #10;
    pin = 4'b1100; #10;
    pin = 4'b0110; #10;
    reset = 1; #10;
    reset = 0; pin = 4'b1111; #10;
    $stop;
  end
endmodule
