module demux_1x4_df(input in, input [1:0] sel, output [3:0] out);
  assign out[0] = in & ~sel[1] & ~sel[0];
  assign out[1] = in & ~sel[1] &  sel[0];
  assign out[2] = in &  sel[1] & ~sel[0];
  assign out[3] = in &  sel[1] &  sel[0];
endmodule

module tb_demux_1x4_df;
  reg in;
  reg [1:0] sel;
  wire [3:0] out;

  demux_1x4_df uut (.in(in), .sel(sel), .out(out));

  initial begin
    in = 1'b1; sel = 2'b00; #10;
    sel = 2'b01; #10;
    sel = 2'b10; #10;
    sel = 2'b11; #10;
    in = 1'b0; sel = 2'b01; #10;
    $stop;
  end
endmodule
