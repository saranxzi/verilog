module siso_shift_reg(input clk, input reset, input sin, output sout);
  reg [3:0] shift_reg;

  assign sout = shift_reg[3];

  always @(posedge clk or posedge reset) begin
    if (reset)
      shift_reg <= 4'b0000;
    else
      shift_reg <= {shift_reg[2:0], sin};
  end
endmodule

module tb_siso_shift_reg;
  reg clk, reset, sin;
  wire sout;

  siso_shift_reg uut (.clk(clk), .reset(reset), .sin(sin), .sout(sout));

  initial begin
    clk = 0;
    forever #5 clk = ~clk;
  end

  initial begin
    reset = 1; sin = 0; #10;
    reset = 0; sin = 1; #10;
    sin = 0; #10;
    sin = 1; #10;
    sin = 1; #10;
    sin = 0; #10;
    $stop;
  end
endmodule
