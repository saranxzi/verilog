module jk_ff_beh(input J, input K, input clk, output reg Q, output Qbar);
  assign Qbar = ~Q;

  always @(posedge clk) begin
    case ({J, K})
      2'b00: Q <= Q;
      2'b01: Q <= 0;
      2'b10: Q <= 1;
      2'b11: Q <= ~Q;
    endcase
  end
endmodule

module tb_jk_ff_beh;
  reg J, K, clk;
  wire Q, Qbar;

  jk_ff_beh uut (.J(J), .K(K), .clk(clk), .Q(Q), .Qbar(Qbar));

  initial begin
    clk = 0;
    forever #5 clk = ~clk;
  end

  initial begin
    J = 0; K = 0; #10;
    J = 1; K = 0; #10;
    J = 0; K = 1; #10;
    J = 1; K = 1; #10;
    J = 1; K = 1; #10;
    $stop;
  end
endmodule
