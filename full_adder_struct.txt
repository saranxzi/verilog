module full_adder (
    input A,    
    input B,    
    input Cin,  
    output Sum,  
    output Cout  
);

wire and1_out, and2_out, xor1_out;

xor xor1 (xor1_out, A, B);
xor xor2 (Sum, xor1_out, Cin);

and and1 (and1_out, A, B);
and and2 (and2_out, xor1_out, Cin);

or or1 (Cout, and1_out, and2_out);

endmodule

module full_adder_tb;

reg A, B, Cin;      
wire Sum, Cout;     

full_adder uut (
    .A(A),
    .B(B),
    .Cin(Cin),
    .Sum(Sum),
    .Cout(Cout)
);

initial begin
    $display("A B Cin | Sum Cout");
    $monitor("%b %b %b |  %b    %b", A, B, Cin, Sum, Cout);

    A = 0; B = 0; Cin = 0; #10;
    A = 0; B = 0; Cin = 1; #10;
    A = 0; B = 1; Cin = 0; #10;
    A = 0; B = 1; Cin = 1; #10;
    A = 1; B = 0; Cin = 0; #10;
    A = 1; B = 0; Cin = 1; #10;
    A = 1; B = 1; Cin = 0; #10;
    A = 1; B = 1; Cin = 1; #10;

    $finish;
end

endmodule
