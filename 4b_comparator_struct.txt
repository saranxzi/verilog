module comparator_4bit(input [3:0] A, input [3:0] B, output A_gt_B, output A_eq_B, output A_lt_B);
  wire [3:0] xnor_out;
  wire [3:0] not_A, not_B;
  wire [3:0] A_gt_B_bits, B_gt_A_bits;
  wire eq1, eq2, eq3;
  
  xnor(xnor_out[3], A[3], B[3]);
  xnor(xnor_out[2], A[2], B[2]);
  xnor(xnor_out[1], A[1], B[1]);
  xnor(xnor_out[0], A[0], B[0]);

  and(A_eq_B, xnor_out[3], xnor_out[2], xnor_out[1], xnor_out[0]);

  not(not_A[3], A[3]);
  not(not_A[2], A[2]);
  not(not_A[1], A[1]);
  not(not_A[0], A[0]);

  not(not_B[3], B[3]);
  not(not_B[2], B[2]);
  not(not_B[1], B[1]);
  not(not_B[0], B[0]);

  and(A_gt_B_bits[3], A[3], not_B[3]);
  and(eq1, xnor_out[3], A[2], not_B[2]);
  and(eq2, xnor_out[3], xnor_out[2], A[1], not_B[1]);
  and(eq3, xnor_out[3], xnor_out[2], xnor_out[1], A[0], not_B[0]);
  or(A_gt_B, A_gt_B_bits[3], eq1, eq2, eq3);

  and(B_gt_A_bits[3], B[3], not_A[3]);
  and(eq1, xnor_out[3], B[2], not_A[2]);
  and(eq2, xnor_out[3], xnor_out[2], B[1], not_A[1]);
  and(eq3, xnor_out[3], xnor_out[2], xnor_out[1], B[0], not_A[0]);
  or(A_lt_B, B_gt_A_bits[3], eq1, eq2, eq3);
endmodule

module tb_comparator_4bit;
  reg [3:0] A, B;
  wire A_gt_B, A_eq_B, A_lt_B;

  comparator_4bit uut (.A(A), .B(B), .A_gt_B(A_gt_B), .A_eq_B(A_eq_B), .A_lt_B(A_lt_B));

  initial begin
    A = 4'b0000; B = 4'b0000; #10;
    A = 4'b1010; B = 4'b0101; #10;
    A = 4'b0101; B = 4'b1010; #10;
    A = 4'b1111; B = 4'b1111; #10;
    A = 4'b0011; B = 4'b0010; #10;
    A = 4'b0001; B = 4'b0011; #10;
    $stop;
  end
endmodule
