module full_subtractor (
    input A,    
    input B,    
    input Bin,  
    output Diff,  
    output Bout  
);

wire xor1_out, and1_out, and2_out;

xor xor1 (xor1_out, A, B);
xor xor2 (Diff, xor1_out, Bin);

and and1 (and1_out, ~A, B);
and and2 (and2_out, xor1_out, Bin);

or or1 (Bout, and1_out, and2_out);

endmodule

module full_subtractor_tb;

reg A, B, Bin;      
wire Diff, Bout;     

full_subtractor uut (
    .A(A),
    .B(B),
    .Bin(Bin),
    .Diff(Diff),
    .Bout(Bout)
);

initial begin
    $display("A B Bin | Diff Bout");
    $monitor("%b %b %b |  %b    %b", A, B, Bin, Diff, Bout);

    A = 0; B = 0; Bin = 0; #10;
    A = 0; B = 0; Bin = 1; #10;
    A = 0; B = 1; Bin = 0; #10;
    A = 0; B = 1; Bin = 1; #10;
    A = 1; B = 0; Bin = 0; #10;
    A = 1; B = 0; Bin = 1; #10;
    A = 1; B = 1; Bin = 0; #10;
    A = 1; B = 1; Bin = 1; #10;

    $finish;
end

endmodule
