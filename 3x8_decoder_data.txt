module decoder3x8 (
    input [2:0] A,
    output [7:0] Y
);

assign Y[0] = ~A[2] & ~A[1] & ~A[0];
assign Y[1] = ~A[2] & ~A[1] & A[0];
assign Y[2] = ~A[2] & A[1] & ~A[0];
assign Y[3] = ~A[2] & A[1] & A[0];
assign Y[4] = A[2] & ~A[1] & ~A[0];
assign Y[5] = A[2] & ~A[1] & A[0];
assign Y[6] = A[2] & A[1] & ~A[0];
assign Y[7] = A[2] & A[1] & A[0];

endmodule

module decoder3x8_tb;

reg [2:0] A;      
wire [7:0] Y;     

decoder3x8 uut (
    .A(A),
    .Y(Y)
);

initial begin
    $display("A | Y");
    $monitor("%b | %b", A, Y);

    A = 3'b000; #10;
    A = 3'b001; #10;
    A = 3'b010; #10;
    A = 3'b011; #10;
    A = 3'b100; #10;
    A = 3'b101; #10;
    A = 3'b110; #10;
    A = 3'b111; #10;

    $finish;
end

endmodule
