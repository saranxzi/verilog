verilog
module decoder2x4 (
    input [1:0] A,
    output [3:0] Y
);

assign Y[0] = ~A[1] & ~A[0];
assign Y[1] = ~A[1] & A[0];
assign Y[2] = A[1] & ~A[0];
assign Y[3] = A[1] & A[0];

endmodule

module decoder2x4_tb;

reg [1:0] A;      
wire [3:0] Y;     

decoder2x4 uut (
    .A(A),
    .Y(Y)
);

initial begin
    $display("A  | Y");
    $monitor("%b | %b", A, Y);

    A = 2'b00; #10;
    A = 2'b01; #10;
    A = 2'b10; #10;
    A = 2'b11; #10;

    $finish;
end

endmodule
