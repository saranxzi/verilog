module full_subtractor (
    input A,    
    input B,    
    input Bin,  
    output reg Diff,  
    output reg Bout  
);

always @ (A or B or Bin) begin
    Diff = A ^ B ^ Bin;    
    Bout = (~A & B) | (Bin & ~(A ^ B));  
end

endmodule

module full_subtractor_tb;

reg A, B, Bin;      
wire Diff, Bout;     

full_subtractor uut (
    .A(A),
    .B(B),
    .Bin(Bin),
    .Diff(Diff),
    .Bout(Bout)
);

initial begin
    $display("A B Bin | Diff Bout");
    $monitor("%b %b %b |  %b    %b", A, B, Bin, Diff, Bout);

    A = 0; B = 0; Bin = 0; #10;
    A = 0; B = 0; Bin = 1; #10;
    A = 0; B = 1; Bin = 0; #10;
    A = 0; B = 1; Bin = 1; #10;
    A = 1; B = 0; Bin = 0; #10;
    A = 1; B = 0; Bin = 1; #10;
    A = 1; B = 1; Bin = 0; #10;
    A = 1; B = 1; Bin = 1; #10;

    $finish;
end

endmodule
